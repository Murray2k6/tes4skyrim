cmake_minimum_required(VERSION 3.25)
project(TES4Runtime VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${SDK_ROOT}/spdlog" spdlog)
file(WRITE "${CMAKE_BINARY_DIR}/packages/spdlogConfig.cmake"
     "set(spdlog_INCLUDE_DIRS \"${SDK_ROOT}/spdlog/include\")\n")
set(spdlog_DIR "${CMAKE_BINARY_DIR}/packages")
set(RAPIDCSV_INCLUDE_DIRS "${SDK_ROOT}/rapidcsv/src")
add_subdirectory("${SDK_ROOT}/CommonLibSSE-NG" CommonLibSSE)
add_library(TES4Runtime SHARED plugin.cpp collections.cpp core.cpp form_types.cpp leveled.cpp references.cpp equipment.cpp inventory.cpp reference_walk.cpp factions.cpp active_effects.cpp script_state.cpp menus.cpp projectiles.cpp merchant.cpp map.cpp voices.cpp events.cpp settings.cpp item_values.cpp magic_items.cpp)
target_link_libraries(TES4Runtime PRIVATE CommonLibSSE::CommonLibSSE)
target_sources(TES4Runtime PRIVATE input.cpp)
target_link_libraries(TES4Runtime PRIVATE user32)
target_compile_definitions(TES4Runtime PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_precompile_headers(TES4Runtime PRIVATE <RE/Skyrim.h> <SKSE/SKSE.h>)
set_target_properties(TES4Runtime PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist")
